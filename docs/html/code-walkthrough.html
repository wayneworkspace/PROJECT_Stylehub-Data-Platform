"""Run the whole pipeline for one date, step by step. Each step is its own function.

Airflow runs the same steps, one task per step (see deploy/airflow/dags/stylehub_daily.py).
Silver and Gold are built by dbt, which also runs the tests, so there is no separate checks step.

Every step is logged, recorded in BigQuery (pipeline_runs) and sends an alert if it fails.

Usage:
    python -m orchestration.run_daily --date 2026-10-01 [--from-step bronze] [--to-step gold]
"""

import argparse
import logging
import sys
import uuid
from collections.abc import Callable
from datetime import UTC, datetime

from ingestion import csv_tables
from orchestration.alerts import send_alert
from orchestration.runs import record_run, run_row
from storage.events_bronze import ensure_events_bronze
from storage.load_bronze import TABLES, load_day
from transform.run_dbt import run_dbt

Module docstring and imports

The docstring says what the file does and how to start it. The runner is the only place that knows the order of the steps, so both the command line and Airflow reuse it. The imports show which parts of the project this file glues together: ingestion from csv_tables, the Bronze loader, the events builder, and dbt. Alerts and run records come from orchestration, and the standard library gives time, ids and logging. Notice that no step does real work in this file. It only calls other modules, which keeps the runner short and easy to read. Because every import is at the top, a missing package shows up as soon as you start the program, not in the middle of a run.

Logger and error limit

log is a named logger called pipeline. Every message from the runner goes through it, so you can see the same lines in the terminal and in Airflow logs. MAX_ERROR_CHARS caps the text of an error at 500 characters. The error goes into one row of the pipeline_runs table, and a huge stack trace would make that row heavy. A named constant with a comment is better than a bare number inside the code, because the reason sits next to the value. If you want longer errors in the history table, this is the only line to change.

def step_ingest(data_date: str) -> int | None:
    # The course data comes from CSV files: full load, no watermark.
    # ingestion/optional/ (Postgres and API) stays for a live source. This flow does not call it.
    for table in csv_tables.CSV_TABLES:
        log.info(csv_tables.ingest_table(table, data_date))
    return None


def step_bronze(data_date: str) -> int | None:
    # events is built once (copy + partitioned table); later runs skip it.
    ensure_events_bronze()
    total = 0
    for table in TABLES:
        rows = load_day(table, data_date)
        log.info("%s: %s rows", table, rows)
        total += rows
    return total


def step_silver(data_date: str) -> int | None:
    run_dbt("silver")
    return None


def step_gold(data_date: str) -> int | None:
    run_dbt("gold")
    return None


# A step takes the data date and returns the rows it loaded, or None if it does not count rows.
STEPS = [
    ("ingest", step_ingest),
    ("bronze", step_bronze),
    ("silver", step_silver),
    ("gold", step_gold),
]

The four step functions and the STEPS list

Each step is a small function that takes the data date and returns the number of rows it loaded, or None when it does not count rows. step_ingest loops over the four CSV tables and runs a full load for each one. The Postgres and API ingestion files are kept, but this step does not call them. step_bronze first makes sure the big events table exists, then loads the latest Raw batch of each table into Bronze and adds up the row counts. step_silver and step_gold only call dbt for one layer. dbt builds the models and runs the tests in one command, so a failed test fails the step. The STEPS list gives each function a name and fixes the order. The names are the same ones that Airflow uses for its task ids and that you pass to --from-step. Because every function has the same shape, run_step can wrap any of them in the same way.

def run_step(
    run_id: str, name: str, func: Callable[[str], int | None], data_date: str
) -> None:
    """Run one step. Record it and alert on failure. Re-raise the error so the run stops."""
    started = datetime.now(UTC)
    status, rows, error = "success", None, None
    log.info("[run %s] step %s starts for %s", run_id, name, data_date)
    try:
        rows = func(data_date)
    except (Exception, SystemExit) as exc:
        status, error = "failed", (str(exc) or type(exc).__name__)[:MAX_ERROR_CHARS]
        log.error("[run %s] step %s FAILED: %s", run_id, name, error)
        raise
    finally:
        finished = datetime.now(UTC)
        record_run(run_row(run_id, name, data_date, status, started, finished, rows, error))
        if status == "failed":
            send_alert(
                f"StyleHub pipeline FAILED\nstep: {name}\ndate: {data_date}\n"
                f"run: {run_id}\nerror: {error}"
            )
    log.info("[run %s] step %s done in %.1fs", run_id, name, (finished - started).total_seconds())

run_step: log, record and alert

This wrapper is the rule of the project: every step goes through it. It notes the start time, then calls the step function inside try. If the step fails, it saves a short error text and raises the error again, so the run stops. The except catches Exception and also SystemExit, because tools such as dbt can stop the program with SystemExit. The finally block always runs, so a success and a failure both write one row to pipeline_runs. Recording and alerts are best effort: they must never crash the pipeline. Only a failed step sends an alert, and the message names the step, the date, the run id and the error. The last log line prints how long the step took. Together with the row in BigQuery, you can see which step is slow without opening Airflow.

def new_run_id() -> str:
    return datetime.now(UTC).strftime("%Y%m%dT%H%M%S") + "-" + uuid.uuid4().hex[:6]


def run_steps(data_date: str, first_step: str, last_step: str) -> None:
    """Run the steps from first_step to last_step (both included) for one data date."""
    names = [name for name, _ in STEPS]
    run_id = new_run_id()
    for name, func in STEPS[names.index(first_step) : names.index(last_step) + 1]:
        run_step(run_id, name, func, data_date)


def setup_logging() -> None:
    logging.basicConfig(
        stream=sys.stdout,
        level=logging.INFO,
        format="%(asctime)s %(levelname)s %(message)s",
    )

Run id, running a range of steps, logging

new_run_id builds a short id from the UTC time and six random letters. It has the same shape as a batch_id, so rows in pipeline_runs are easy to sort. run_steps picks the slice of STEPS from the first to the last step (both included) and runs each one through run_step. All steps in one call share the same run id. Airflow calls run_steps with the same name for the first and last step, so one Airflow task runs exactly one step. setup_logging sends log lines to standard output with a time stamp. It is a separate function, because Airflow's Python task must call it by hand. Without it, the INFO lines would be lost.

def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--date", required=True, help="YYYY-MM-DD")
    parser.add_argument("--from-step")
    parser.add_argument("--to-step")
    args = parser.parse_args()

    names = [name for name, _ in STEPS]
    from_step, to_step = args.from_step or names[0], args.to_step or names[-1]
    for step in (from_step, to_step):
        if step not in names:
            parser.error(f"step '{step}' does not exist: {names}")
    if names.index(from_step) > names.index(to_step):
        parser.error("--from-step must come before --to-step")

    setup_logging()
    run_steps(args.date, from_step, to_step)


if __name__ == "__main__":
    main()

Command line entry point

main reads three options: a required --date, and optional --from-step and --to-step. With no step options it runs all four steps. It checks the input before it starts any work: a step name that does not exist, or a first step that comes after the last one, stops the program with a clear message. Then it sets up logging and calls run_steps. That is all, because the real logic lives in the functions above. Running the same date twice is safe, because each step is idempotent: Bronze replaces the batch of that day and Silver keeps the newest row of each id. A typical use is --from-step silver --to-step gold after you change a business rule: it rebuilds the tables without reading the source again.

def extract_full(table: str):
    """Every row of one CSV table, read from the BigQuery dataset with the CSV data."""
    if table not in CSV_TABLES:
        raise ValueError(f"No CSV table configured: {table}")
    cfg = settings()
    client = bigquery.Client(project=cfg.project)
    query = f"SELECT * FROM `{cfg.project}.{cfg.source}.{table}`"
    df = call_with_retry(lambda: client.query(query).to_dataframe())
    return df.drop(columns=DROP_COLUMNS.get(table, []), errors="ignore")

extract_full: read a whole table

First the function checks that the table name is one of the allowed names. A typo then fails with a clear message and does not run a wrong query. It builds a BigQuery client from the project settings and runs SELECT * on project.source.table. The names come from settings(), so nothing is hard-coded. The query runs inside call_with_retry, so a short BigQuery or network error is tried again with growing waits. Other errors, such as a missing table, fail at once. to_dataframe() turns the result into a pandas table. The tables are small (at most 180 thousand rows), so one pass in memory is fine. The 108 million row events table does not use this path. Finally it drops the columns listed in DROP_COLUMNS. errors="ignore" means a table without such columns passes through unchanged. The data goes out unchanged, apart from this one drop. That follows the rule that Raw holds the data as fetched.

def ingest_table(
    table: str,
    data_date: str,
    dry_run: bool = False,
    on_schema_change: OnChange = "fail",
) -> str:
    """Extract one CSV table. With dry_run, only count the rows and write nothing to GCS."""
    load_env()
    started = time.perf_counter()
    df = enforce_schema(extract_full(table), table, on_schema_change)

    if dry_run:
        seconds = time.perf_counter() - started
        note = f"csv, {seconds:.1f}s, dry run, nothing written"
        return f"{table} {data_date}: {len(df)} rows ({note})"
    uri = write_raw(df, table, data_date, source=SOURCE_NAME)
    seconds = time.perf_counter() - started
    return f"{table} {data_date}: {len(df)} rows (csv, {seconds:.1f}s) -> {uri}"

ingest_table: check the columns and write Raw

This function is the whole step for one table. It loads the settings file, starts a timer, reads the table and runs enforce_schema. enforce_schema compares the real columns with the expected ones. By default a missing or changed column stops the run (fail), so a silent change in the source cannot reach Bronze. With dry_run it only counts rows and writes nothing to GCS. That is a cheap way to test the connection and the schema. Without it, write_raw saves one Parquet file under raw/stylehub/<table>/date=<date>/batch_id=<id>/. It adds _ingested_at, _source and _batch_id. The function returns one text line with the table, the date, the row count, the time and the file address. The runner prints that line in the log.

"""Load the Raw files of one day into Bronze. Safe to re-run.

Each ingest writes a new batch under the day's Raw folder. Bronze keeps only the LATEST batch of
each day, so running a day twice does not double the rows. Different days stay side by side, so
Bronze still holds one snapshot per day and Silver picks the newest row of each id.

The order is: load the new batch first, delete the older batches after. If the load fails, the old
rows are still there, so Bronze never loses a day. If the delete fails, the day has two batches
for a while. That is harmless (Silver keeps the newest row) and the next run removes the old one.

Usage: python -m storage.load_bronze --date 2026-10-01 [--table orders]
"""

import argparse

from google.api_core.exceptions import NotFound
from google.cloud import bigquery, storage

from common.config import RAW_DIR, settings
from common.retry import call_with_retry

The docstring also gives the order of work: load the new batch first, delete the older ones after. def batch_ids_from_names(names: list[str]) -> list[str]:
    """The batch ids found in Raw file names, oldest first (ids start with a timestamp)."""
    ids = set()
    for name in names:
        for part in name.split("/"):
            if part.startswith("batch_id="):
                ids.add(part.removeprefix("batch_id="))
    return sorted(ids)

batch_ids_from_names: find batches in file names

A Raw file path looks like raw/stylehub/orders/date=2026-10-01/batch_id=20261001T020000-ab12cd/orders.parquet. This function reads the part that starts with batch_id=. It collects every batch_id in a set, so several files of one batch count once, and returns the ids sorted. An id starts with a UTC time stamp, so a text sort puts the oldest batch first and the newest last. The function does not touch the network. That makes it easy to test with a plain list of strings.

def list_batch_ids(prefix: str) -> list[str]:
    """The batch ids of one table and one day, read from the Raw Zone on GCS."""
    cfg = settings()
    client = storage.Client(project=cfg.project)
    # The listing is lazy, so read it inside the retried function.
    names = call_with_retry(
        lambda: [blob.name for blob in client.list_blobs(cfg.bucket, prefix=prefix)]
    )
    return batch_ids_from_names(names)

list_batch_ids: ask GCS

The function lists all files under one prefix, which means one table and one day, in the Raw bucket. It passes the file names to batch_ids_from_names and returns the sorted batch ids. The listing from the Storage client is lazy: nothing is fetched until you loop over it. So the loop sits inside the function given to call_with_retry. A short network error during the listing is then retried, not raised. An empty list means that no ingest has run for that table and date. The caller treats this as nothing to load. Listing is cheap, because it reads names and not file contents.

def table_exists(client: bigquery.Client, table_id: str) -> bool:
    try:
        call_with_retry(lambda: client.get_table(table_id))
    except NotFound:
        return False
    return True


def count_batch_rows(client: bigquery.Client, table_id: str, batch_id: str) -> int:
    """How many rows of a batch are already in Bronze."""
    query = f"SELECT COUNT(*) AS n FROM `{table_id}` WHERE _batch_id = @batch_id"
    params = [bigquery.ScalarQueryParameter("batch_id", "STRING", batch_id)]
    config = bigquery.QueryJobConfig(query_parameters=params)
    rows = call_with_retry(lambda: client.query(query, job_config=config).result())
    return next(iter(rows)).n


def delete_batches(client: bigquery.Client, table_id: str, batch_ids: list[str]) -> None:
    """Remove the rows of older batches of the same day."""
    query = f"DELETE FROM `{table_id}` WHERE _batch_id IN UNNEST(@batch_ids)"
    params = [bigquery.ArrayQueryParameter("batch_ids", "STRING", batch_ids)]
    config = bigquery.QueryJobConfig(query_parameters=params)
    call_with_retry(lambda: client.query(query, job_config=config).result())

BigQuery helpers: exists, count, delete

table_exists asks BigQuery for the table, through call_with_retry, and returns False when it is missing. NotFound is not a short error, so it is not retried: it reaches the except at once. On the very first run the Bronze table does not exist yet. count_batch_rows counts the rows with one _batch_id. It uses a query parameter, not text pasted into the SQL, so the value cannot break the query. delete_batches removes the rows of older batches with a DELETE statement and IN UNNEST(@batch_ids). Rows loaded by a load job can be deleted at once, so this works without the delay that streaming inserts have. Both queries also run inside call_with_retry. These three helpers are small on purpose. load_day reads like a short story because they hide the BigQuery details.

def load_day(table: str, data_date: str) -> int:
    """Make Bronze hold the latest Raw batch of one table and one day. Return the rows loaded.

    Returns 0 when nothing was loaded: no Raw file, or the latest batch is already in Bronze.
    """
    cfg = settings()
    raw_dir = RAW_DIR.get(table, table)
    prefix = f"{cfg.raw_prefix}/{raw_dir}/date={data_date}/"
    table_id = f"{cfg.project}.{cfg.bronze}.{table}"

    batch_ids = list_batch_ids(prefix)
    if not batch_ids:
        return 0
    latest, older = batch_ids[-1], batch_ids[:-1]

    client = bigquery.Client(project=cfg.project)
    already_loaded = table_exists(client, table_id) and count_batch_rows(client, table_id, latest)

    rows = 0
    if not already_loaded:
        config = bigquery.LoadJobConfig(
            source_format=bigquery.SourceFormat.PARQUET,
            write_disposition=bigquery.WriteDisposition.WRITE_APPEND,
        )
        uri = f"gs://{cfg.bucket}/{prefix}batch_id={latest}/*"

        def load() -> int:
            job = client.load_table_from_uri(uri, table_id, job_config=config)
            job.result()
            return job.output_rows or 0

        rows = call_with_retry(load)

    if older:  # only after the new batch is safe in Bronze
        delete_batches(client, table_id, older)
    return rows

load_day: load the new batch first, delete the old one after

First it lists the batches of this table and day. If there are none, it returns 0 and does nothing. The newest id is the one to load. All older ids are leftovers from earlier runs of the same day. If the Bronze table exists and already holds the newest batch, the function skips the load. Otherwise it starts a load job from that one batch folder with WRITE_APPEND. The load runs inside call_with_retry, so a short BigQuery error starts a new job instead of failing the step. Only after the new batch is safe in Bronze does the function delete the older batches. This order is the key point: if the load fails, the old rows are still there and Bronze never loses a day. If the delete fails, the day has two batches for a while. That is harmless, because Silver keeps the newest row of each id, and the next run removes the old batch. The result is idempotent: a second run of a day finds the newest batch already loaded, loads nothing, and only cleans up older batches if any are left. A new ingest on the same day creates a new batch. The next load adds it and removes the old one, so Bronze never keeps two copies of one day for long. The function returns the number of rows loaded. The runner adds these numbers and stores the total in pipeline_runs.

def run_python_step(data_date, step, project_home):

    Airflow copies the source of this function to the other Python, so it must not use anything
    defined outside it. Imports go inside.

    sys.path.insert(0, f"{project_home}/src/python")

    run_daily.run_steps(data_date, step, step)

run_python_step: the function Airflow copies

Airflow does not import this function in the other Python. It copies its source text to that environment and runs it there, so it must be self-contained. That is why import sys sits inside the function. This is how an ExternalPythonOperator works. The first line adds src/python to sys.path, so from orchestration import run_daily works. Then it calls setup_logging(). Without that call the INFO lines of the step would be lost, because the other process has no logging set up. The last line runs exactly one step through run_steps, so the record in pipeline_runs and the alert still happen. The function has no type hints and no outside names on purpose: anything defined outside it would be missing in the other Python.

    dag_id="stylehub_daily",
    description="StyleHub: ingest, load Bronze, build Silver and Gold with dbt",
    default_args=default_args,
    start_date=datetime(2026, 10, 1),
    # 02:00 every day. A run at 02:00 on the 15th has ds = the 14th: yesterday's data.
    schedule="0 2 * * *",
    # Old days are loaded on purpose with: airflow dags backfill (see deploy/airflow/README.md)
    catchup=False,
    # Silver and Gold are rebuilt as a whole, so two runs must never overlap.
    max_active_runs=1,
    tags=["stylehub"],
) as dag:
    start = EmptyOperator(task_id="start")
    end = EmptyOperator(task_id="end")

    tasks = []
    for step in STEP_NAMES:
        if step in PYTHON_STEPS:
            task = ExternalPythonOperator(
                task_id=step,
                python=PIPELINE_PYTHON,
                python_callable=run_python_step,
                op_kwargs={"data_date": "{{ ds }}", "step": step, "project_home": PROJECT_HOME},
                # The pipeline environment has no Airflow and no pendulum.
                expect_airflow=False,
                expect_pendulum=False,
            )
        else:
            task = BashOperator(task_id=step, bash_command=dbt_command(step))
        tasks.append(task)

    # start >> ingest >> bronze >> silver >> gold >> end
    start >> tasks[0]
    for upstream, downstream in zip(tasks, tasks[1:]):
        upstream >> downstream
    tasks[-1] >> end
